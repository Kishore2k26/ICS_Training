4.) Projects

create table Projects (
    Project_ID int primary key,
    Descr varchar(30) not null,
    StartDate date,
    Planned_End_Date date,
    Actual_End_Date date,
    Budget decimal(10,0) check (Budget > 0),
    Client_ID int,
    constraint CK_Project_EndDate
        check (Actual_End_Date > Planned_End_Date OR Actual_End_Date is null),
    constraint FK_Projects_Clients
        foreign key (CLient_ID)
        references Clients(CLient_ID)

);

5.) Project_Employees (EmpProjectTasks)

create table Project_Employees (
    Project_ID int not null,
    Empno int not null,
    StartDate date,
    End_Date date,
    Task varchar(25) not null,
    PE_Status varchar(15) not null check (PE_Status in ('In Progress', 'Completed', 'Cancelled')),

    constraint PK_Project_Employees
        primary key (Project_ID, Empno),

    constraint FK_PE_Project
        foreign key (Project_ID)
        references Projects(Project_ID),

    constraint FK_PE_Employee
        foreign key (Empno)
        references Employees(Empno)
);
